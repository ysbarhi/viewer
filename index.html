function setupFileUploader() { const fileBtn = document.getElementById('file-btn'); const fileInput = document.getElementById('file-input'); const closeProjectBtn = document.getElementById('close-project-btn'); // --- NOUVEAU : Récupération du fichier via le menu "Partager" d'Android --- const urlParams = new URLSearchParams(window.location.search); if (urlParams.get('shared') === 'true') { caches.open('shared-file-cache').then(cache => { cache.match('/shared-file.glb').then(response => { if (response) { response.blob().then(blob => { const objectUrl = URL.createObjectURL(blob); loadModelFromUrl(objectUrl); cache.delete('/shared-file.glb'); // Nettoyage de la mémoire window.history.replaceState({}, document.title, "./index.html"); // Nettoyage de l'URL }); } }); }); } // Ouverture PC (File Handlers) if ('launchQueue' in window) { window.launchQueue.setConsumer(async (launchParams) => { if (launchParams.files && launchParams.files.length > 0) { const fileHandle = launchParams.files[0]; const file = await fileHandle.getFile(); const objectUrl = URL.createObjectURL(file); loadModelFromUrl(objectUrl); } }); } // Ouverture manuelle via le bouton fileBtn.addEventListener('click', () => fileInput.click()); fileInput.addEventListener('change', function(event) { const file = event.target.files[0]; if (file) { const objectUrl = URL.createObjectURL(file); loadModelFromUrl(objectUrl); } }); closeProjectBtn.addEventListener('click', () => { if (globalModel) { scene.remove(globalModel); globalModel = null; } document.getElementById('start-menu').style.display = 'flex'; showUIElements(false); fileInput.value = ""; document.getElementById('explode-slider').value = 0; document.getElementById('texture-btn').classList.remove('active'); document.getElementById('xray-btn').classList.remove('active'); isTextureOn = false; isXrayOn = false; baseMaterial.opacity = 1.0; edgeMaterial.opacity = 1.0; }); }
